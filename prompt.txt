from .state import State
from typing import Literal
from langchain_core.messages import AIMessage
from utils.logger import get_logger_with_session
import json
import re
import ast

NodeType = Literal["Planner", "Manager", "Coder", "Reporter", "Cartographer", "CodeReviewer", "MapReviewer", "Recorder",  "Refiner"]
ProcessNodeType = Literal['Coder', 'Reporter', 'Manager', 'Cartographer']

def process_router(state: State) -> ProcessNodeType:
    logger = get_logger_with_session()
    logger.info("Entering process_router")
    
    process_decision = state.get("process_decision", "")
    
    logger.debug(f"process_decision: {process_decision}")
    logger.debug(f"process_decision_type: {type(process_decision)}")

    if len(state["messages"]) >= 20:
        logger.info("Messages length >= 20. Ending process.")
        return "END"
    
    if isinstance(process_decision, AIMessage):
        logger.debug("Process decision is an AIMessage")
        try:

            content = re.sub(r"(?<!\w)'(?!\w)", '"', process_decision.content)
            decision_dict = json.loads(content)
            process_decision = decision_dict.get('next', '')
            logger.debug(f"Parsed process decision from JSON: {process_decision}")
        except json.JSONDecodeError:
            process_decision = process_decision.content
            logger.warning("Failed to parse process decision as JSON. Using content directly.")
    elif isinstance(process_decision, dict):
        process_decision = process_decision.get('next', '') 
        logger.debug(f"Process decision is a dictionary. Using 'next' value: {process_decision}")
    elif not isinstance(process_decision, str):
        process_decision = str(process_decision)
        logger.warning(f"Unexpected process decision type. Converting to string: {process_decision}")
    
    if isinstance(process_decision, str):
        try:
            decision_dict = json.loads(process_decision.replace("'", '"'))
            process_decision = decision_dict.get('next', '')
            logger.debug(f"Parsed process decision from JSON: {process_decision}")
        except json.JSONDecodeError:
            decision_dict = ast.literal_eval(process_decision)
            process_decision = decision_dict.get('next', '')
            logger.debug(f"Parsed process decision from dictionary: {process_decision}")

    valid_decisions = {"Coder", "Reporter", "Cartographer"}
    
    if process_decision in valid_decisions:
        logger.info(f"Valid process decision: {process_decision}")
        return process_decision
    
    if process_decision == "FINISH":
        logger.info("Process decision is FINISH. Ending process.")

        return "END"
    
    if not process_decision or process_decision not in valid_decisions:
        logger.warning(f"Invalid or empty process decision: {process_decision}. Defaulting to 'Manager'.")
        return "Manager"
    
    logger.info("Defaulting to 'Manager'")
    return "Manager"

def quality_review_router(state: State) -> NodeType:
    logger = get_logger_with_session()
    logger.info("Entering quality review router")
    
    messages = state.get("messages", [])
    last_message = messages[-1] if messages else None
    
    if (last_message and 'REVISION' in str(last_message.content)) or state.get("needs_revision", False):
        current_node = state.get("sender", "")
        revision_routes = {
            "codeReviewer": "Coder", 
            "mapReviewer": "Cartographer",
            "reportReviewer": "Reporter", 
        }
        result = revision_routes.get(current_node, "Recorder")
        logger.info(f"Revision needed. Routing to: {result}")
        return result
    
    else:
        return "Recorder"

@tool
def get_sub_categories_by_category(main_category: str) -> list[str]:
    """
    Get subcategories list by main category
    
    Args:
        main_category (str): Main category name, e.g. "1 Geometric and Related Operations"
        
    Returns:
        list[str]: List of subcategories, returns error message if main category does not exist
    """
    logger = get_logger_with_session()
    logger.info(f"Reading subcategories: main_category: {main_category}")

    # Read catalog configuration file
    catalogue = read_yaml("../book/Geospatial_Analysis/catalogue.yaml")

    # catalogue is a dictionary, first check if main_category exists as a key
    if main_category not in catalogue:
        return f"Main category does not exist, please check input parameters: main_category: {main_category}"

    # Get all subcategories under the main category
    sub_categories = catalogue[main_category]

    # Return subcategories
    return sub_categories

@tool
def get_book_content(main_category: str, sub_category: str) -> str:
    """
    Get content based on main category and subcategory name
    
    Args:
        main_category: Main category name with index, e.g. "1 Geometric and Related Operations"
        sub_category: Subcategory name with index, e.g. "1.1 Length and area for vector data"
        
    Returns:
        str: Content text, truncated if exceeds 5000 characters
    """
    logger = get_logger_with_session()
    # Build markdown file path
    logger.info(f"Reading book content: main_category: {main_category}, sub_category: {sub_category}")
    markdown_path = f"../book/Geospatial_Analysis/{main_category}/{sub_category}.md"
    
    # Check if file exists
    if not os.path.exists(markdown_path):
        return f"File does not exist, please check input parameters: main_category: {main_category} sub_category: {sub_category}"
    
    # Load markdown file
    loader = UnstructuredMarkdownLoader(markdown_path)
    data = loader.load()
    
    # Get content
    content = data[0].page_content
    
    # Truncate if exceeds 5000 characters
    if len(content) > 5000:
        content = content[:5000]
        
    return content

@tool

@tool
def query_function_document(
    function_name: Annotated[str, "The name of the function to query, typically from the GIS library."]
) -> Annotated[str, "The documentation string for the specified function or an error message if retrieval fails."]:
    """
    Query the documentation of a function.

    This function retrieves the documentation string of a specified function by its name. 
    It is especially useful for GIS functions like `sjoin` to understand their usage.
    Avoid abbreviations for libraries and use the full name. (e.g. use geopandas instead of gpd, use pandas.DataFrame.to_csv instead of pd.DataFrame.to_csv)
    Only query the function name, not the class name. (e.g. geopandas.GeoDataFrame is not a valid function name)

    Args:
        function_name (str): The name of the function whose documentation is to be retrieved.
            The function should be in the current namespace or importable.

    Returns:
        str: The documentation string for the specified function. If the function is not found or an error occurs,
        an error message is returned.

    Examples:
        Example 1: Query documentation for `sjoin`
        >>> query_function_document("geopandas.sjoin") 
        # Output:
        # Documentation for sjoin:
        # Perform spatial join between GeoDataFrames...
    """
    logger = get_logger_with_session()
    logger.info(f"(query_function_document) Querying documentation for function: {function_name}")
    try:
        # Split module name and function name
        module_name, func_name = function_name.rsplit('.', 1)
        
        # Dynamically import module
        module = import_module(module_name)
        func = getattr(module, func_name)
        
        # Use inspect to get documentation string
        documentation = inspect.getdoc(func)
        if not documentation:
            documentation = "No documentation available."
        
        return f"Documentation for {function_name}:\n{documentation}"
    
    except ValueError:
        logger.error("(query_function_document) Invalid function name format. Please use format 'module.function'")
        return "Error: Invalid function name format. Please use format 'module.function'"
    except ImportError:
        logger.error(f"(query_function_document) Module {module_name} not found")

    """
    Read data from a CSV file and return a summary of the table, trying multiple encodings if necessary.

    Args:
        data_path (str): The path to the CSV file.
        
    Returns:
        dict: A dictionary containing the summary of the CSV data.
        
    Raises:
        FileNotFoundError: If the file doesn't exist
        ValueError: If file is empty or too large
        RuntimeError: If file cannot be read with any encoding
    """
    logger = get_logger_with_session()
    
    working_directory = os.environ.get("WORKING_DIRECTORY", os.getcwd())
    if working_directory not in data_path:
        data_path = os.path.abspath(os.path.join(working_directory, data_path))
    
    if not os.path.exists(data_path):
        logger.error(f"(reading_csv_summary) File not found: {data_path}")
        raise FileNotFoundError(f"File not found: {data_path}")
        
    file_size = os.path.getsize(data_path)
    if file_size == 0:
        logger.error("File is empty")
        raise ValueError("File is empty")
    if file_size > 1_000_000_000:  # 1GB
        logger.error("File is too large (>1GB)")
        raise ValueError("File is too large (>1GB)")
    
    encodings = ['utf-8', 'utf-8-sig', 'latin1', 'iso-8859-1', 'cp1252', 'gbk', 'big5']
    errors = {}
    
    logger.info(f"(reading_csv_summary) Attempting to read CSV file: {data_path}")
    
    for encoding in encodings:
        try:
            data = pd.read_csv(data_path, encoding=encoding)
            if len(data.columns) > 0: 
                logger.info(f"(reading_csv_summary) Successfully read CSV file with encoding: {encoding}")
                
                summary = {
                    'encoding': encoding,
                    'num_rows': len(data),
                    'num_columns': len(data.columns),
                    'column_names': list(data.columns),
                    'column_types': data.dtypes.apply(lambda x: str()).to_dict(),
                    'sample_data': data.head(3).to_dict(orient='records')
                }
                return summary
            else:
                raise ValueError("No columns found in the CSV file")
        except Exception as e:
            errors[encoding] = str(e)
            logger.debug(f"(reading_csv_summary) Failed with encoding {encoding}: {e}")
            continue
    
    error_msg = "Unable to read file with any encoding. Errors:\n" + \
                "\n".join([f"{enc}: {err}" for enc, err in errors.items()])
    logger.error(error_msg)
    raise RuntimeError(error_msg)

                centroid_x = centroid.x.mean()
                centroid_y = centroid.y.mean()
                spatial_variance = centroid.distance(centroid[0]).mean()
                
                if vector_data.crs:
                    centroid_points = gpd.GeoSeries([centroid[0]], crs=3857).to_crs(vector_data.crs)
                    centroid_x = centroid_points.x[0]
                    centroid_y = centroid_points.y[0]
            else:
                centroid = vector_data.geometry.centroid
                centroid_x = centroid.x.mean()
                centroid_y = centroid.y.mean()
                spatial_variance = centroid.distance(centroid[0]).mean()

@tool
def save_file_metadata(role: str, file_name: str, file_path: str, description: str):
    """
    This function saves metadata about a file generated by a specific role.
    If `file_metadata.json` does not exist, it creates the file and adds the entry.
    It supports multiple files per role and accepts the description of each file as a required parameter.

    Args:
        role (str): The role (e.g., 'User', 'Planner', 'Coder', 'Cartographer', 'Reporter'). Capitalize the first letter of the role.
        file_name (str): The name of the file being saved.
        file_path (str): The full path of the file.
        description (str): A brief description of the file's purpose or role.
    """
    logger = get_logger_with_session()

    # Get the working directory from environment variable
    working_path = os.environ.get("WORKING_DIRECTORY", "")
    if not working_path:
        raise EnvironmentError("(save_file_metadata) WORKING_DIRECTORY environment variable is not set.")

    # Ensure the directory exists
    if not os.path.exists(working_path):
        os.makedirs(working_path)

    # Full path for the metadata file
    metadata_file_path = os.path.join(working_path, "file_metadata.json")

    # Define a lock for the metadata file
    lock = FileLock(metadata_file_path + ".lock")

    # Try to acquire the lock and perform file operations
    with lock:
        try:
            # Check if metadata file exists, if not, create it
            if os.path.exists(metadata_file_path):
                with open(metadata_file_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
            else:
                # Initialize the metadata structure
                data = {
                    "metadata": {
                        "last_updated": datetime.utcnow().isoformat(),
                        "roles": []
                    },
                    "files": {}
                }
        except json.JSONDecodeError:
            logger.error(f"(save_file_metadata) Invalid JSON format in {metadata_file_path}. Creating a new file.")
            data = {
                "metadata": {
                    "last_updated": datetime.utcnow().isoformat(),
                    "roles": []
                },
                "files": {}
            }

        # Ensure the role exists in the file structure
        if role not in data["files"]:
            data["files"][role] = []

        # Prepare the file entry
        file_metadata = {
            "file_name": file_name,
            "file_path": file_path,
            "description": description,
            "timestamp": datetime.utcnow().isoformat(),
        }

        # Append the file metadata to the role's list if not already added
        if not any(file['file_name'] == file_name for file in data["files"][role]):
            data["files"][role].append(file_metadata)

        # Update the last_updated timestamp
        data["metadata"]["last_updated"] = datetime.utcnow().isoformat()

        # Check if the role exists in the roles list, add it if not
        if role not in data["metadata"]["roles"]:
            data["metadata"]["roles"].append(role)

        # Save the updated metadata to the file
        try:
            with open(metadata_file_path, 'w', encoding='utf-8') as f:
                json.dump(data, f, ensure_ascii=False, indent=4)
            logger.info(f"(save_file_metadata) File metadata for {role} -> {file_name} has been saved.")
        except Exception as e:
            logger.error(f"(save_file_metadata) Error saving metadata: {e}")

class CustomFormatter(logging.Formatter):

    reset = "\x1b[0m"          
    
    format = "%(asctime)s - %(name)s - %(levelname)s - %(message)s"

    FORMATS = {
        "DEFAULT": grey + format + reset,
        "Init": blue + format + reset,
        "Agent": green + format + reset,
        "Node": magenta + format + reset,
        "Router": cyan + format + reset,
        "Tool": orange + format + reset,
        "Main": {
            logging.INFO: orange + format + reset,
            logging.WARNING: yellow + format + reset,
            logging.ERROR: red + format + reset,
            logging.CRITICAL: bold_red + format + reset,
            logging.DEBUG: cyan + format + reset
        },
        logging.WARNING: yellow + format + reset,
        logging.ERROR: red + format + reset,
        logging.CRITICAL: bold_red + format + reset
    }

    def format(self, record):
        session_id = os.environ.get("SESSION_ID", "default")
        
        if session_id in record.name:
            log_fmt = self.FORMATS["Main"].get(record.levelno, self.FORMATS["DEFAULT"])
        else:
            log_fmt = self.FORMATS.get(record.levelno, self.FORMATS["DEFAULT"])
        
        formatter = logging.Formatter(log_fmt, "%Y-%m-%d %H:%M:%S")
        return formatter.format(record)

def setup_logger_with_session_id(session_id: str):
    
    logger = logging.getLogger(session_id)

        return logger

    log_dir = Path("logs")

    file_handler = logging.FileHandler(log_file, mode='a', encoding='utf-8')
    file_formatter = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s', "%Y-%m-%d %H:%M:%S")
    file_handler.setFormatter(file_formatter)
    
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(CustomFormatter())
    
    logger.setLevel(logging.INFO)
    logger.addHandler(file_handler)
    logger.addHandler(console_handler)
    logger.propagate = False

def copy_files(src_paths: Union[str, List[str]], dst_dir: str) -> None:

        os.makedirs(dst_dir)
    
    if isinstance(src_paths, str):
        src_paths = [src_paths]

def get_subtask_number(dir_name: str) -> int:
    """从 'subtask_N' 提取数字 N，以便进行自然排序"""
    match = re.search(r'\d+', dir_name)
    return int(match.group()) if match else 0

# 核心函数：运行 Agent 并返回指定 schema 的 JSON
# ==========================================
def run_agent_to_json(task_num: str, task: str, data_path: list) -> str:
    """
    执行 run_agent 并将生成的代码和图片结果打包为 JSON 字符串返回。
    返回的 JSON 结构:
    {
        "code": "[合并的 Python 代码]",
        "images": ["[图片的 base64 编码字符串1]", "[图片的 base64 编码字符串2]", ...]
    }
    """
    logger.info(f"Executing run_agent_to_json for task_num: {task_num}")


    working_directory = os.environ["WORKING_DIRECTORY"]

    # 提取 source（取第一个元素解析即可，因为都相同）
    first_item = data_path[0]
    source = first_item.split('/')[0]  # "sfgpt"
    base_dir = Path("datasets/Datasets") / source

    # 用于存放最终结果的列表（或使用集合去重）
    data_paths_set = set()

    for item in data_path:
        # 解析 prefix 和 format
        filename = item.split('/')[-1]  # "roads_1.shp" / "line.shp"
        prefix, ext = os.path.splitext(filename)  # prefix="roads_1", ext=".shp"
        fmt = ext[1:] if ext.startswith('.') else ext  # "shp"

        # 在目标目录下匹配所有 prefix.fmt 文件
        for p in base_dir.glob(f"{prefix}.*"):
            data_paths_set.add(str(p))

    # 转回列表（如需保持顺序，可改用 list 并手动判断重复）
    data_paths = list(data_paths_set)

    print(data_paths)

    # 1. 运行 agent 获取最终回复与工作目录
    final_message = run_agent(task_num, task, data_paths)
    logger.info(f"Agent finished. Working dir: {working_directory}")

    work_dir_path = Path(working_directory)
    if not work_dir_path.exists() or not work_dir_path.is_dir():
        logger.error(f"Working directory not found: {working_directory}")
        raise RuntimeError("Working directory not found after execution.")

    # 2. 收集并拼接 coder 目录下的所有 Python 代码
    coder_dir = work_dir_path / "coder"
    all_python_code = []

    if coder_dir.exists():
        # 获取所有 subtask_N 目录并按数字排序
        subtasks = [d for d in coder_dir.iterdir() if d.is_dir() and d.name.startswith("subtask_")]
        subtasks.sort(key=lambda d: get_subtask_number(d.name))

        for subtask_dir in subtasks:
            for file_path in subtask_dir.iterdir():
                # 收集 .py 文件内容并拼接
                if file_path.is_file() and file_path.suffix.lower() == ".py":
                    try:
                        code_content = file_path.read_text(encoding='utf-8')
                        all_python_code.append(
                            f"# {'=' * 40}\n# File: {subtask_dir.name}/{file_path.name}\n# {'=' * 40}\n\n{code_content}\n"
                        )
                    except Exception as e:
                        logger.warning(f"Failed to read python file {file_path}: {e}")

    concatenated_code = "\n".join(all_python_code) if all_python_code else "# No python code found.\n"

    # 3. 收集 cartographer 目录下的图片并将其转换为 Base64
    cartographer_dir = work_dir_path / "cartographer"
    image_extensions = {".jpg", ".jpeg", ".png", ".gif", ".bmp", ".tif", ".tiff", ".svg"}
    images_base64 = []

    if cartographer_dir.exists():
        for root, _, files in os.walk(cartographer_dir):
            for file_name in files:
                file_path = Path(root) / file_name
                if file_path.suffix.lower() in image_extensions:
                    try:
                        with open(file_path, "rb") as img_file:
                            encoded_string = base64.b64encode(img_file.read()).decode('utf-8')
                            images_base64.append(encoded_string)
                    except Exception as e:
                        logger.warning(f"Failed to encode image file {file_path}: {e}")

    # 4. 构建并返回指定的 JSON 结构
    result_dict = {
        "code": concatenated_code,
        "images": images_base64
    }

    logger.info("Successfully assembled JSON response.")
    return json.dumps(result_dict, ensure_ascii=False)
